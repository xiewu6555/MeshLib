## 概述

MeshLib 采用混合依赖管理策略，结合 vcpkg 自动管理和手动管理，确保所有依赖的正确配置和版本控制。

### 依赖管理架构

```
MeshLib 依赖管理
├── vcpkg 管理的依赖（自动）
│   ├── Boost 库集合
│   ├── 图形库（GLFW、OpenGL）
│   ├── 图像处理（libjpeg、libpng、tiff）
│   ├── 数据格式（jsoncpp、tinyxml2）
│   └── 专业库（E57、GDCM、OpenVDB）
├── 手动管理的依赖
│   ├── OpenCASCADE（CAD 几何）
│   └── CUDA Toolkit（GPU 加速）
└── Git 子模块
    ├── parallel-hashmap
    └── clip
```

### 依赖分类

| 类别 | 管理方式 | 示例 | 说明 |
|------|----------|------|------|
| **核心依赖** | vcpkg | Boost、Eigen3、TBB | 必需的基础库 |
| **可选依赖** | vcpkg/手动 | OpenCASCADE、GDCM | 特定功能支持 |
| **开发依赖** | vcpkg | GoogleTest、spdlog | 测试和调试 |
| **子模块** | Git | parallel-hashmap | 直接集成的代码 |

## vcpkg 依赖管理

### 项目级 vcpkg 配置

MeshLib 使用项目独立的 vcpkg 实例，避免系统级污染。

#### 配置文件结构

```

├── vcpkg.json              # 依赖清单
├── vcpkg-configuration.json # vcpkg 配置

├── setup-vcpkg.sh          # Linux/macOS 初始化脚本
└── vcpkg/                  # 本地 vcpkg 安装（初始化后生成）
    ├── vcpkg.exe
    ├── scripts/
    ├── triplets/
    └── installed/
```

### vcpkg 操作命令

#### 基础操作

```cmd
# 初始化项目 vcpkg
setup-vcpkg.bat

# 安装所有依赖
.\vcpkg\vcpkg.exe install

# 安装特定架构
.\vcpkg\vcpkg.exe install --triplet x64-windows

# 安装带特性的依赖
.\vcpkg\vcpkg.exe install --triplet x64-windows --x-feature=opencascade,python

# 更新依赖
.\vcpkg\vcpkg.exe upgrade

# 查看已安装的包
.\vcpkg\vcpkg.exe list

# 搜索可用包
.\vcpkg\vcpkg.exe search boost

# 删除包
.\vcpkg\vcpkg.exe remove boost-algorithm:x64-windows

# 清理缓存
.\vcpkg\vcpkg.exe remove --outdated
```

#### 高级操作

```cmd
# 导出已安装的包
.\vcpkg\vcpkg.exe export --zip --output=meshlib-deps.zip

# 集成到系统（不推荐，使用项目级配置）
.\vcpkg\vcpkg.exe integrate install

# 查看包信息
.\vcpkg\vcpkg.exe info eigen3

# 检查依赖关系
.\vcpkg\vcpkg.exe depend-info boost-algorithm:x64-windows

# 使用二进制缓存
set VCPKG_BINARY_SOURCES=clear;files,F:\vcpkg-cache,readwrite
.\vcpkg\vcpkg.exe install
```

### 自定义 Triplet

创建 `vcpkg/triplets/community/x64-windows-meshlib.cmake`：

```cmake
# 基础配置
set(VCPKG_TARGET_ARCHITECTURE x64)
set(VCPKG_CRT_LINKAGE dynamic)
set(VCPKG_LIBRARY_LINKAGE dynamic)

# 编译器选项
set(VCPKG_C_FLAGS "/DWIN32 /D_WINDOWS /W3 /utf-8")
set(VCPKG_CXX_FLAGS "/DWIN32 /D_WINDOWS /W3 /utf-8 /EHsc /std:c++20")

# Release 配置
set(VCPKG_C_FLAGS_RELEASE "/MD /O2 /Ob2 /DNDEBUG")
set(VCPKG_CXX_FLAGS_RELEASE "/MD /O2 /Ob2 /DNDEBUG")

# Debug 配置
set(VCPKG_C_FLAGS_DEBUG "/MDd /Od /Ob0 /RTC1")
set(VCPKG_CXX_FLAGS_DEBUG "/MDd /Od /Ob0 /RTC1")

# 特定包的配置
if(PORT MATCHES "opencascade")
    set(VCPKG_BUILD_TYPE "release")
endif()

if(PORT MATCHES "openvdb")
    set(VCPKG_CXX_FLAGS "${VCPKG_CXX_FLAGS} /bigobj")
endif()
```

## 手动依赖管理

### OpenCASCADE 手动安装

#### 下载和安装

1. **获取 OpenCASCADE 7.9.1**：
   - 官网：https://old.opencascade.com/content/previous-releases
   - 直接链接：opencascade-7.9.1-vc14-64.zip

2. **目录结构**：
```
thirdparty/opencascade/
└── occt_vc14-64/
    ├── win64/
    │   └── vc14/
    │       ├── bin/      # 187 个 DLL 文件
    │       ├── lib/      # 静态库和导入库
    │       └── inc/      # 头文件
    ├── samples/          # 示例代码
    └── doc/              # 文档
```

3. **CMake 集成模块**：

`cmake/Modules/UseManualOpenCASCADE.cmake`：
```cmake
# UseManualOpenCASCADE.cmake
# 手动安装的 OpenCASCADE 配置

if(NOT USE_MANUAL_OPENCASCADE)
    return()
endif()

message(STATUS "==============================================")
message(STATUS "配置手动安装的 OpenCASCADE")
message(STATUS "==============================================")

# 设置根目录
set(OPENCASCADE_ROOT "${CMAKE_SOURCE_DIR}/thirdparty/opencascade/occt_vc14-64")

# 验证安装
if(NOT EXISTS "${OPENCASCADE_ROOT}")
    message(FATAL_ERROR "未找到 OpenCASCADE：${OPENCASCADE_ROOT}")
endif()

# 设置路径
set(OpenCASCADE_INCLUDE_DIR "${OPENCASCADE_ROOT}/inc")
set(OpenCASCADE_LIBRARY_DIR "${OPENCASCADE_ROOT}/win64/vc14/lib")
set(OpenCASCADE_BINARY_DIR "${OPENCASCADE_ROOT}/win64/vc14/bin")

# 核心库列表
set(OpenCASCADE_CORE_LIBRARIES
    TKernel
    TKMath
    TKG2d
    TKG3d
    TKGeomBase
    TKBRep
    TKGeomAlgo
    TKTopAlgo
    TKPrim
    TKBO
    TKBool
    TKShHealing
    TKMesh
    TKFillet
    TKOffset
)

# 数据交换库
set(OpenCASCADE_DATAEXCHANGE_LIBRARIES
    TKSTEP
    TKSTEP209
    TKSTEPAttr
    TKSTEPBase
    TKIGES
    TKSTL
    TKVRML
    TKXSBase
    TKXCAF
    TKXDEIGES
    TKXDESTEP
)

# 可视化库
set(OpenCASCADE_VISUALIZATION_LIBRARIES
    TKV3d
    TKService
    TKOpenGl
)

# 创建接口目标
add_library(OpenCASCADE::OpenCASCADE INTERFACE IMPORTED)

# 设置包含目录
target_include_directories(OpenCASCADE::OpenCASCADE INTERFACE
    ${OpenCASCADE_INCLUDE_DIR}
)

# 收集所有库
set(OpenCASCADE_ALL_LIBRARIES)
foreach(lib ${OpenCASCADE_CORE_LIBRARIES} ${OpenCASCADE_DATAEXCHANGE_LIBRARIES})
    if(EXISTS "${OpenCASCADE_LIBRARY_DIR}/${lib}.lib")
        list(APPEND OpenCASCADE_ALL_LIBRARIES "${OpenCASCADE_LIBRARY_DIR}/${lib}.lib")
    endif()
endforeach()

# 链接库
target_link_libraries(OpenCASCADE::OpenCASCADE INTERFACE
    ${OpenCASCADE_ALL_LIBRARIES}
)

# 定义预处理器宏
target_compile_definitions(OpenCASCADE::OpenCASCADE INTERFACE
    HAVE_OPENCASCADE
    _USE_MATH_DEFINES
    WNT
)

# 收集 DLL 文件
file(GLOB OpenCASCADE_DLLS "${OpenCASCADE_BINARY_DIR}/*.dll")
set(OPENCASCADE_RUNTIME_DLLS ${OpenCASCADE_DLLS} CACHE INTERNAL "")

# 创建拷贝 DLL 的函数
function(copy_opencascade_dlls TARGET_NAME)
    if(WIN32 AND OPENCASCADE_RUNTIME_DLLS)
        foreach(dll ${OPENCASCADE_RUNTIME_DLLS})
            add_custom_command(TARGET ${TARGET_NAME} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${dll}"
                "$<TARGET_FILE_DIR:${TARGET_NAME}>"
            )
        endforeach()
        
        message(STATUS "配置了 ${TARGET_NAME} 的 OpenCASCADE DLL 拷贝")
    endif()
endfunction()

# 设置变量
set(OpenCASCADE_FOUND TRUE)
set(OpenCASCADE_VERSION "7.9.1")

# 输出配置信息
message(STATUS "OpenCASCADE 配置成功")
message(STATUS "  版本: ${OpenCASCADE_VERSION}")
message(STATUS "  包含目录: ${OpenCASCADE_INCLUDE_DIR}")
message(STATUS "  库目录: ${OpenCASCADE_LIBRARY_DIR}")
message(STATUS "  DLL 目录: ${OpenCASCADE_BINARY_DIR}")
message(STATUS "  核心库数量: ${CMAKE_SIZEOF_VOID_P}")
list(LENGTH OPENCASCADE_RUNTIME_DLLS dll_count)
message(STATUS "  DLL 文件数量: ${dll_count}")
message(STATUS "==============================================")
```

### CUDA Toolkit 配置

#### 安装和配置

1. **下载 CUDA Toolkit**：
   - https://developer.nvidia.com/cuda-downloads
   - 版本：11.4 或更高

2. **CMake 配置**：
```cmake
# 启用 CUDA
enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)

# 设置架构
set(CMAKE_CUDA_ARCHITECTURES "75;86")

# 创建 CUDA 目标
add_library(MRCuda SHARED
    ${CUDA_SOURCES}
)

target_link_libraries(MRCuda
    CUDA::cudart
    CUDA::cublas
    CUDA::cusparse
)

# 设置 CUDA 编译选项
target_compile_options(MRCuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:
        --use_fast_math
        --relocatable-device-code=true
        --extended-lambda
    >
)
```

## 依赖冲突解决

### 常见冲突类型

1. **ABI 不兼容**
   - 症状：链接错误，未定义符号
   - 解决：确保所有依赖使用相同的编译器和设置

2. **版本冲突**
   - 症状：API 不匹配，编译错误
   - 解决：使用 overrides 固定版本

3. **运行时冲突**
   - 症状：DLL 版本错误
   - 解决：使用依赖隔离和正确的加载顺序

### 诊断工具

```cmd
# 检查依赖关系
.\vcpkg\vcpkg.exe depend-info boost:x64-windows

# 查看 DLL 依赖
dumpbin /dependents MeshViewer.exe

# 使用 Dependencies Walker
Dependencies.exe MeshViewer.exe
```

## 最佳实践

### 1. 依赖管理原则

- **最小依赖原则**：只添加必需的依赖
- **版本一致性**：使用 baseline 确保版本一致
- **定期更新**：定期更新依赖但要测试兼容性
- **文档化**：记录特殊依赖的配置要求

### 2. 性能优化

```cmake
# 使用静态链接减少 DLL 依赖（适用于部署）
set(VCPKG_LIBRARY_LINKAGE static)

# 使用二进制缓存加速安装
set(VCPKG_BINARY_SOURCES "clear;files,F:/vcpkg-cache,readwrite")

# 并行安装
set(VCPKG_MAX_CONCURRENCY 8)
```

### 4. 故障恢复

创建 `reset_dependencies.bat`：

```batch
@echo off
echo 重置项目依赖...

:: 备份当前配置
xcopy /y vcpkg.json vcpkg.json.backup

:: 清理 vcpkg
rmdir /s /q vcpkg_installed
rmdir /s /q vcpkg\downloads
rmdir /s /q vcpkg\packages
rmdir /s /q vcpkg\buildtrees

:: 重新初始化
call setup-vcpkg.bat

echo 依赖重置完成
```
